import express from "express"; import cors from "cors"; import * as cheerio from "cheerio"; import { URL } from "node:url"; const app = express(); const PORT = process.env.PORT || 10000; app.use(cors({origin:true})); app.use(express.json({limit:"2mb"})); // ========================= // HELPERS // ========================= function absolute(link, base){ try{ return new URL(link, base).href; }catch{ return null; } } function cleanPhone(value){ const p = String(value || "") .replace(/\D/g,""); if(p.length >= 10 && p.length <= 15){ return p; } return ""; } function cleanEmail(value){ const e = String(value || "") .trim() .toLowerCase(); if( e.includes("@") && e.includes(".") ){ return e; } return ""; } async function getPage(url){ try{ const response = await fetch(url,{ headers:{ "User-Agent": "Mozilla/5.0" } }); const html = await response.text(); return html; }catch{ return ""; } } // ========================= // ENCONTRAR PERFIS // ========================= function extractProfiles(html,base){ const $ = cheerio.load(html); let links=[]; $("a[href]").each((_,el)=>{ let href = $(el).attr("href"); if(!href) return; if( href.startsWith("#") || href.startsWith("javascript") ){ return; } const url = absolute( href, base ); if(!url) return; // remove lixo const bad=[ "login", "entrar", "termos", "politica", "contato", "facebook", "instagram", "youtube", "javascript" ]; if( bad.some(x => url.toLowerCase() .includes(x)) ){ return; } // pega somente paginas com slug const path = new URL(url).pathname; const parts = path.split("/") .filter(Boolean); if(parts.length >= 1){ links.push(url); } }); return [ ...new Set(links) ]; } // ========================= // EXTRAI PERFIL // ========================= function extractData(html,url){ const $ = cheerio.load(html); let nome = $("h1") .first() .text() .trim(); if(!nome){ nome = $("title") .text() .trim(); } let whatsapp=""; let telefone=""; let email=""; $("a").each((_,el)=>{ const href = $(el) .attr("href") || ""; if( href.includes("wa.me") || href.includes("whatsapp") ){ whatsapp = cleanPhone(href); } if( href.startsWith("tel:") ){ telefone = cleanPhone( href.replace("tel:","") ); } if( href.startsWith("mailto:") ){ email = cleanEmail( href.replace("mailto:","") ); } }); const texto = $("body") .text() .replace(/\s+/g," "); if(!telefone){ const phones = texto.match( /(?:\+55\s?)?\(?\d{2}\)?\s?\d{4,5}[- ]?\d{4}/g ); if(phones){ telefone = cleanPhone( phones[0] ); } } if(!email){ const emails = texto.match( /[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}/ig ); if(emails){ email = cleanEmail( emails[0] ); } } return { nome: nome || "Não encontrado", whatsapp, telefone, email, urlPerfil:url }; } // ========================= // ROTAS // ========================= app.get("/",(req,res)=>{ res.json({ status:"online", message: "Lead Collector AI API funcionando" }); }); app.get("/health",(req,res)=>{ res.json({ online:true, service:"Lead Collector AI" }); }); // ========================= // COLETOR // ========================= app.post("/api/collect",async(req,res)=>{ try{ const site = String(req.body.url || "") .trim(); if(!site){ return res.status(400) .json({ error:"Informe uma URL" }); } const home = await getPage(site); if(!home){ return res.json({ status:"error", message:"Site inacessível" }); } const profiles = extractProfiles( home, site ); let leads=[]; // máximo inicial 100 for( const profile of profiles.slice(0,100) ){ const html = await getPage(profile); if(html){ const data = extractData( html, profile ); leads.push(data); } } res.json({ status:"success", totalPerfisEncontrados: profiles.length, processados: leads.length, temMais: profiles.length > 100, leads }); }catch(err){ res.status(500) .json({ error: err.message }); } }); app.listen(PORT,()=>{ console.log( `Lead Collector AI rodando ${PORT}` ); });
