import express from "express"; import cors from "cors"; import * as cheerio from "cheerio"; import { URL } from "node:url"; const app = express(); const PORT = process.env.PORT || 10000; app.use(cors({origin:true})); app.use(express.json({limit:"1mb"})); // =============================== // NORMALIZADORES // =============================== function cleanPhone(value){ const phone = String(value || "") .replace(/\D/g,""); if(phone.length >= 10 && phone.length <= 15){ return phone; } return ""; } function cleanEmail(value){ const email = String(value || "") .trim() .toLowerCase(); if(email.includes("@")){ return email; } return ""; } function absolute(url,base){ try{ return new URL(url,base).href; } catch{ return null; } } // =============================== // BAIXAR PAGINA // =============================== async function fetchHTML(url){ try{ const response = await fetch(url,{ headers:{ "User-Agent": "Mozilla/5.0 LeadCollectorAI" } }); const type = response.headers.get("content-type") || ""; if(!type.includes("text/html")){ return ""; } return await response.text(); }catch{ return ""; } } // =============================== // EXTRAIR PERFIS // =============================== function findProfileLinks(html,base){ const $ = cheerio.load(html); const links=[]; $("a[href]").each((_,el)=>{ const href = absolute( $(el).attr("href"), base ); if(!href) return; if( href !== base && href.includes("/") ){ links.push(href); } }); return [...new Set(links)]; } // =============================== // EXTRAIR DADOS DO PERFIL // =============================== function extractProfile(html,url){ const $ = cheerio.load(html); let name=""; // tenta pegar titulo name = $("h1").first().text().trim() || $("title").text().trim(); let phones=[]; $("a[href^='tel:']") .each((_,el)=>{ const p = cleanPhone( $(el) .attr("href") .replace("tel:","") ); if(p) phones.push(p); }); let emails=[]; $("a[href^='mailto:']") .each((_,el)=>{ const e = cleanEmail( $(el) .attr("href") .replace("mailto:","") ); if(e) emails.push(e); }); const text = $("body") .text() .replace(/\s+/g," "); for(const m of text.matchAll( /(?:\+?55)?\s*\(?\d{2}\)?\s*\d{4,5}[-\s]?\d{4}/g )){ const p = cleanPhone(m[0]); if(p) phones.push(p); } for(const m of text.matchAll( /[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}/ig )){ const e = cleanEmail(m[0]); if(e) emails.push(e); } return { name:name || "Não identificado", whatsapp: phones[0] || "", phone: phones.join(", "), email: [...new Set(emails)].join(", "), url:url }; } // =============================== // ROTAS // =============================== app.get("/",(req,res)=>{ res.json({ status:"online", message: "Lead Collector AI API funcionando" }); }); app.get("/health",(req,res)=>{ res.json({ online:true, service:"Lead Collector AI" }); }); // =============================== // COLETOR PRINCIPAL // =============================== app.post("/api/collect",async(req,res)=>{ try{ const target = String(req.body.url || "") .trim(); if(!target){ return res.status(400) .json({ error:"Informe uma URL" }); } const startHTML = await fetchHTML(target); if(!startHTML){ return res.json({ status:"error", message: "Não foi possível acessar o site" }); } const profiles = findProfileLinks( startHTML, target ); let results=[]; const maxProfiles = 100; for( const profile of profiles.slice(0,maxProfiles) ){ const html = await fetchHTML(profile); if(html){ const data = extractProfile( html, profile ); results.push(data); } } res.json({ status:"success", pagesFound: profiles.length, profilesProcessed: results.length, hasMore: profiles.length > maxProfiles, contacts:results }); } catch(error){ res.status(500) .json({ error:error.message }); } }); app.listen(PORT,()=>{ console.log( `API online na porta ${PORT}` ); });
